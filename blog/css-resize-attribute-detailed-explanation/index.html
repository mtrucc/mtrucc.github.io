      div.bar {
        width: 300px;
        height: 100%;
        margin-right: -2px;
        /* 因为 bar 盒子和 text 盒子是重合的所以必须设置一个负值才能看到我们的拖动按钮*/
        overflow: scroll;
        /* 可设置其他值来进行对别，只可意会，不可言传*/
        resize: horizontal;
        /*水平拖放*/
      }

      /*因为拖动按钮属于滚动条系列，所以可通过改变滚动来间接改变拖动按钮的样式*/
      div.bar::-webkit-scrollbar {
        width: 10px;
        height: 9999px;
        /*可设为 100% 或者其他数值来查看展示效果，只可意会，不可言传*/
      }

      div.bar::-webkit-resizer {
        /* 设置拖放按钮的背景颜色 */
        background: skyblue;
      }
